//   derived key (20 bytes)  conn+0x118
//   inbound cipher state    conn+0x12C  (i@0, j@4, S[256]@8)

// ---------------------------------------------------------------------------
// 1. Dump the transport key material (this is what unlocks a raw capture).
// ---------------------------------------------------------------------------
savedata [mod.base("Gw2-64.exe")+0x26632d0]+0x118, "C:/gw2capture/conn118.bin", 0x14
savedata [mod.base("Gw2-64.exe")+0x26632d0]+0x12C, "C:/gw2capture/conn12C.bin", 0x108

// ---------------------------------------------------------------------------
// 2. Optional: dump each decoded message's wire bytes.
//
//    Hardware execution breakpoint at MsgPack_ReadFields entry. On a hit:
//       RDX -> cursor variable; *RDX = start of the current message
//       R8  -> payloadEnd
//    so the message wire bytes are [*RDX, R8). Run this in the command bar on
//    the first hit, then F9 to continue, or wire it as the breakpoint command:
//
//       savedata [rdx], "C:/gw2capture/msg.bin", r8-[rdx]
//
//    Hardware breakpoints do not modify code. Prefer them over software INT3.
// ---------------------------------------------------------------------------
bphws mod.base("Gw2-64.exe")+0xfebc30, "x"

// Logs the base so the session can be tied to the image in the manifest.
log "gw2 capture dumps requested; base={mod.base(\"Gw2-64.exe\")}"
